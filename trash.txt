
def compute_cartesian_harmonic_force(coords, coords0, masses_au, cnorm, omega2):
    """
    Harmonic force in Cartesian coordinates.

    Parameters
    ----------
    coords : (natm, 3) ndarray
        Current Cartesian coordinates in Bohr.
    coords0 : (natm, 3) ndarray
        Equilibrium Cartesian coordinates in Bohr.
    masses_au : (natm, 1) or (natm,) ndarray
        Atomic masses in atomic units (electron masses).
    cnorm : (nmodes, nmodes) ndarray
        Mass-weighted normal-mode matrix.
    omega2 : (nmodes,) ndarray
        Squared frequencies in atomic units.
        Rot/trans modes must already be set to zero.

    Returns
    -------
    f_cart : (natm, 3) ndarray
        Harmonic Cartesian force in Hartree/Bohr.
    """

    masses_au = np.asarray(masses_au, dtype=float).reshape(-1, 1)
    sqrtm = np.sqrt(masses_au)

    dx = np.asarray(coords, dtype=float) - np.asarray(coords0, dtype=float)

    # mass-weighted Cartesian displacement
    dq_cart = sqrtm * dx

    # normal-mode coordinates
    q = cnorm.T @ dq_cart.ravel()

    # harmonic force in normal-mode space
    f_mode = -omega2 * q

    # back to Cartesian force
    f_cart = (sqrtm * (cnorm @ f_mode).reshape(coords.shape))

    return f_cart
